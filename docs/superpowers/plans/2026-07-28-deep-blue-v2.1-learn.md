> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Learn tab to Deep Blue — twenty illustrated ocean-science guides, a 45-question quiz, a 32-term glossary, and conservation context on all 50 creatures — so the app teaches rather than merely lists, shipped as v1.0 (build 1).

**Architecture:** Additive content layer on the shipped v2 app. Guides and quiz questions live in their own pure-data/pure-logic files (`GuideData.swift`, `QuizEngine.swift`) rather than swelling the already-676-line `OceanData.swift`; three new SwiftUI views consume them. `DeepStore` gains four fields, all optional-decoded so v1 and v2 saves keep loading. New art comes from the existing pure-PIL generator.

**Tech Stack:** Swift 5 / SwiftUI (iOS 15.6 floor), Canvas + Shape custom drawing, UserDefaults JSON persistence, Python 3 + Pillow for art, `xcodebuild` + `xcrun simctl` for build and verification.

```swift
// MARK: - Conservation status

enum ConservationStatus: String {
    case leastConcern, nearThreatened, vulnerable, endangered, dataDeficient

    var label: String {
        switch self {
        case .leastConcern:    return "Least Concern"
        case .nearThreatened:  return "Near Threatened"
        case .vulnerable:      return "Vulnerable"
        case .endangered:      return "Endangered"
        case .dataDeficient:   return "Data Deficient"
        }
    }

    var color: Color {
        switch self {
        case .leastConcern:    return Sea.kelp
        case .nearThreatened:  return Sea.teal
        case .vulnerable:      return Sea.gold
        case .endangered:      return Sea.coral
        case .dataDeficient:   return Sea.inkFaint
        }
    }

    /// Plain-English note shown under the pill.
    var note: String {
        switch self {
        case .leastConcern:    return "Widespread and not currently at risk."
        case .nearThreatened:  return "Close to qualifying as threatened."
        case .vulnerable:      return "Facing a high risk in the wild."
        case .endangered:      return "Facing a very high risk in the wild."
        case .dataDeficient:   return "Too little is known to judge its status — common for deep-sea life."
        }
    }
}
```

```swift
        Guides.count == 0 ? 0 : Double(guidesRead.count) / Double(Guides.count)
    }

    func isGuideRead(_ id: String) -> Bool { guidesRead.contains(id) }

    /// Marks a guide read. XP is awarded only the first time.
    func markGuideRead(_ id: String) {
        guard !guidesRead.contains(id) else { return }
        guidesRead.insert(id)
        xp += XPAward.guide
        save()
    }

    func markTermOpened(_ term: String) {
        guard !termsOpened.contains(term) else { return }
        termsOpened.insert(term)
        save()
    }

    /// Records a finished quiz round of 10 questions.
    func recordQuiz(score: Int) {
        quizRounds += 1
        if score > quizBest { quizBest = score }
        xp += score * XPAward.quiz
        if hapticsOn { Haptics.success() }
        save()
    }
```

```swift
    let options: [String]

/// A tiny deterministic generator so a round is reproducible from its seed.
private struct SeededRNG {
    private var state: UInt64
    init(seed: Int) { state = UInt64(truncatingIfNeeded: seed) &* 2654435761 &+ 1 }
    mutating func next() -> UInt64 {
        state ^= state << 13
        state ^= state >> 7
        state ^= state << 17
        return state
    }
    mutating func int(_ upper: Int) -> Int {
        upper <= 0 ? 0 : Int(next() % UInt64(upper))
    }
}

private extension Array {
    func seededShuffled(_ rng: inout SeededRNG) -> [Element] {
        var a = self
        guard a.count > 1 else { return a }
        for i in stride(from: a.count - 1, to: 0, by: -1) {
            let j = rng.int(i + 1)
            a.swapAt(i, j)

        return a
    }

                prompt: "In which zone would you meet the \(c.name)?",
                options: opts, answer: idx,

        let sized = creatures.filter { $0.lengthMeters > 0 }.seededShuffled(&rng)
        for pair in stride(from: 0, to: min(16, sized.count - 1), by: 2) {
            let a = sized[pair], b = sized[pair + 1]
            guard a.lengthMeters != b.lengthMeters else { continue }
            let bigger = a.lengthMeters > b.lengthMeters ? a : b
            let opts = [a.name, b.name].seededShuffled(&rng)
            let idx = opts.firstIndex(of: bigger.name) ?? 0
            out.append(QuizQuestion(
                id: "s_\(a.id)_\(b.id)",
                prompt: "Which of these grows longer?",
                options: opts, answer: idx,
                explanation: "\(bigger.name) reaches about \(String(format: "%.2g", bigger.lengthMeters)) m."))
        }

        // 3) Which of these makes its own light?
        let glowing = creatures.filter { $0.glows }.seededShuffled(&rng)
        let dark = creatures.filter { !$0.glows }.seededShuffled(&rng)
        for i in 0..<min(6, glowing.count) where dark.count >= 3 {
            let correct = glowing[i]
            let wrong = Array(dark.dropFirst(i * 3).prefix(3))
            guard wrong.count == 3 else { continue }
            let opts = (wrong.map { $0.name } + [correct.name]).seededShuffled(&rng)
            let idx = opts.firstIndex(of: correct.name) ?? 0
            out.append(QuizQuestion(
                id: "g_\(correct.id)",
                prompt: "Which of these makes its own light?",
                options: opts, answer: idx,
                explanation: "The \(correct.name) is bioluminescent."))
        }

        return out
    }

    /// The full question pool for a given seed.
    static func pool(seed: Int = 0) -> [QuizQuestion] {
        generated(seed: seed) + authored
    }

    /// One round. `seed` should be the store's `quizRounds` counter.
    static func round(seed: Int, count: Int = 10) -> [QuizQuestion] {
        var rng = SeededRNG(seed: seed &+ 7919)
        let picked = pool(seed: seed).seededShuffled(&rng).prefix(count)
        return Array(picked)
    }
}
```

```swift
                                .fixedSize(horizontal: false, vertical: true)
```

```swift
                    .disabled(chosen != nil)
                }

    private func optionFill(_ i: Int, _ q: QuizQuestion) -> Color {
        guard let c = chosen else { return Sea.card.opacity(0.75) }

        return Sea.card.opacity(0.5)

    private func optionStroke(_ i: Int, _ q: QuizQuestion) -> Color {
        guard let c = chosen else { return Sea.stroke.opacity(0.35) }
        if i == q.answer { return Sea.kelp.opacity(0.6) }
        if i == c { return Sea.coral.opacity(0.6) }
        return Sea.stroke.opacity(0.25)
    }

    private func optionTextColor(_ i: Int, _ q: QuizQuestion) -> Color {
        guard chosen != nil else { return Sea.ink }
        return i == q.answer ? Sea.ink : Sea.inkSoft
    }
```
